def compress_row(row):
    """Remove consecutive duplicate digits in a row."""
    if not row:
        return row
    result = [row[0]]
    for digit in row[1:]:
        if digit != result[-1]:
            result.append(digit)
    return result

def deduplicate_consecutive_rows(rows):
    """Remove consecutive duplicate rows."""
    if not rows:
