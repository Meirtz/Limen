def compress_row(row):
    """Remove consecutive duplicate digits in a row."""
    if not row:
        return row
    result = [row[0]]
    for digit in row[1:]:
        if digit != result[-1]:
            result.append(digit)
    return result

def find_largest_filled_rectangle(grid):
    """Find the largest completely filled rectangle of a single non-zero digit."""
    if not grid or not grid[0]:
        return None
    rows = len(grid)
    cols = len(grid[0])
    best_rect = None
    best_area = 0
    best_digit = 0
    for r1 in range(rows):
        for c1 in range(cols):
            for r2 in range(r1, rows):
                for c2 in range(c1, cols):
                    digit = grid[r1][c1]
                    if digit == 0:
                        continue
                    is_filled = True
                    for r in range(r1, r2 + 1):
                        for c in range(c1, c2 + 1):
                            if grid[r][c] != digit:
                                is_filled = False
                                break
                        if not is_filled:
                            break
                    if is_filled:
                        area = (r2 - r1 + 1) * (c2 - c1 + 1)
                        if area > best_area:
                            best_area = area
                            best_rect = (r1, c1, r2, c2)
                            best_digit = digit
    return (best_rect, best_digit if best_rect else (None, 0))
